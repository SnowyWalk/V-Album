CREATE TABLE `group` (
	`group_uuid` CHAR(36) NOT NULL COLLATE 'utf8_general_ci',
	`name` VARCHAR(64) NOT NULL COLLATE 'utf8_general_ci',
	`pic` VARCHAR(64) NULL DEFAULT NULL COLLATE 'utf8_general_ci',
	`created_at` DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
	`updated_at` DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
	`deleted_at` DATETIME NULL DEFAULT NULL,
	PRIMARY KEY (`group_uuid`) USING BTREE,
	UNIQUE INDEX `group_uuid` (`group_uuid`) USING BTREE
)
COLLATE='utf8_general_ci'
ENGINE=InnoDB
;
